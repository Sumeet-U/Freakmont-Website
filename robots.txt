# Reminder for future edits: this is a request, not a lock. Well-behaved
# crawlers (Google, Bing, etc.) honor it; it does nothing to actually
# restrict who can load a URL directly. Don't rely on it to keep anything
# truly private -- see legal.html's Data & Privacy section for what
# actually is and isn't exposed.

User-agent: *
# Raw data files aren't useful search results and would just clutter an
# index with JSON -- keep crawlers on the actual pages instead.
Disallow: /data/
# player.html works fine with no query string (shows a blank/prompt
# state) and with one per character (?name=X) -- without this, a crawler
# can "discover" 30 near-duplicate indexed URLs for one page. Let it
# index the page itself; keep it off the per-character variants.
Disallow: /player.html?*
Allow: /
